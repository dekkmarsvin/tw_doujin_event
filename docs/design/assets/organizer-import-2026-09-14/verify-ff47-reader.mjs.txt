import { createHash } from "node:crypto";
import { readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import { createServer, isRunnableDevEnvironment } from "vite";

function parseArguments(arguments_) {
  const options = {};
  for (let index = 0; index < arguments_.length; index += 1) {
    const argument = arguments_[index];
    if (argument === "--workspace" || argument === "--output" || argument === "--label"
      || argument === "--before" || argument === "--after") {
      const value = arguments_[index + 1];
      if (!value) throw new Error(`Missing value for ${argument}.`);
      options[argument.slice(2)] = value;
      index += 1;
    } else if (argument === "--compare") {
      options.compare = true;
    } else {
      throw new Error(`Unknown argument ${argument}.`);
    }
  }
  return options;
}

const options = parseArguments(process.argv.slice(2));
const hashBytes = (bytes) => createHash("sha256").update(bytes).digest("hex");
const readJson = async (filePath) => JSON.parse(await readFile(filePath, "utf8"));

if (options.compare) {
  if (!options.before || !options.after) throw new Error("Comparison requires --before and --after.");
  const before = await readJson(path.resolve(options.before));
  const after = await readJson(path.resolve(options.after));
  const checks = [
    ["pin", JSON.stringify(before.pin) === JSON.stringify(after.pin)],
    ["pinnedInputHashes", JSON.stringify(before.pinnedInputHashes) === JSON.stringify(after.pinnedInputHashes)],
    ["stagedOutputHashes", JSON.stringify(before.stagedOutputHashes) === JSON.stringify(after.stagedOutputHashes)],
    ["map", JSON.stringify(before.map) === JSON.stringify(after.map)],
    ["catalog", JSON.stringify(before.catalog) === JSON.stringify(after.catalog)],
    ["official", JSON.stringify(before.official) === JSON.stringify(after.official)],
    ["projectionCounts", JSON.stringify(before.projectionCounts) === JSON.stringify(after.projectionCounts)],
    ["projectionSha256", before.projectionSha256 === after.projectionSha256],
  ];
  const report = { result: checks.every(([, match]) => match) ? "PASS" : "FAIL", checks };
  if (options.output) await writeFile(path.resolve(options.output), `${JSON.stringify(report, null, 2)}\n`);
  console.log(JSON.stringify(report, null, 2));
  if (report.result !== "PASS") process.exitCode = 1;
} else {
  if (!options.workspace || !options.output) throw new Error("Verification requires --workspace and --output.");
  const workspace = path.resolve(options.workspace);
  const dataRoot = path.join(workspace, ".event-data", "ff47");
  const publicRoot = path.join(workspace, "public", "data", "events", "ff47");
  const pin = await readJson(path.join(workspace, "data", "event-data-pins", "ff47.json"));
  const pinPath = (filePath) => filePath.startsWith("events/ff47/")
    ? path.join(dataRoot, filePath.slice("events/ff47/".length))
    : path.join(dataRoot, filePath);
  const pinnedInputHashes = {};
  for (const file of pin.files) {
    const actual = hashBytes(await readFile(pinPath(file.path)));
    pinnedInputHashes[file.path] = { expected: file.sha256, actual, match: actual === file.sha256 };
  }
  const stagedOutputHashes = {};
  for (const fileName of ["event.json", "circles.json", "map.json", "reference-records.json"]) {
    stagedOutputHashes[fileName] = hashBytes(await readFile(path.join(publicRoot, fileName)));
  }
  const official = await readJson(path.join(dataRoot, "official-booths.json"));
  const officialDays = official.days.map((day) => ({
    day: day.day,
    groups: day.booths.length,
    codes: day.booths.reduce((total, group) => total + group.codes.length, 0),
    multiCodeGroups: day.booths.filter((group) => group.codes.length > 1).length,
  }));

  const vite = await createServer({
    configFile: false,
    root: workspace,
    server: { middlewareMode: true },
    appType: "custom",
    environments: { ssr: {} },
    logLevel: "silent",
  });
  try {
    const environment = vite.environments.ssr;
    if (!isRunnableDevEnvironment(environment)) throw new Error("Vite SSR test environment is not runnable.");
    const { parseEventDefinition } = await environment.runner.import("/app/event-catalog.ts");
    const { buildCircleCatalog, isCircleCatalogPayload } = await environment.runner.import("/app/circle-records.ts");
    const { projectEventWorkspace } = await environment.runner.import("/app/event-workspace-projection.ts");
    const { validateEventMapLayout } = await environment.runner.import("/app/event-map.ts");
    const eventValue = await readJson(path.join(publicRoot, "event.json"));
    const referencesValue = await readJson(path.join(publicRoot, "reference-records.json"));
    const catalogValue = await readJson(path.join(publicRoot, "circles.json"));
    const mapValue = await readJson(path.join(publicRoot, "map.json"));
    const event = parseEventDefinition(eventValue, referencesValue);
    if (!isCircleCatalogPayload(catalogValue)) throw new Error("Staged circles.json is not a valid circle catalog.");
    const catalog = buildCircleCatalog(catalogValue, undefined, event);
    const mapValidation = validateEventMapLayout(mapValue.layout);
    if (!mapValidation.ok) throw new Error(`FF47 map validation failed: ${mapValidation.errors.join("; ")}`);
    const record = (value) => ({
      id: value.id,
      recordId: value.recordId,
      code: value.code,
      name: value.name,
      pen: value.pen,
      genre: value.genre,
      tags: value.tags,
      day: value.day,
      hall: value.hall,
      x: value.x,
      y: value.y,
      tone: value.tone,
      work: value.work,
      note: value.note,
      sources: value.sources,
      circle: value.circle,
      placement: value.placement,
    });
    const planning = { schemaVersion: 3, favoriteGroups: [], favorites: [], visitPlans: [] };
    const base = {
      event,
      records: catalog.records,
      recordsById: catalog.recordsById,
      recordsByCircleId: catalog.recordsByCircleId,
      planning,
      area: "ALL",
      genre: event.genres[0],
      query: "",
      favoriteOnly: false,
      advancedSearch: {
        creatorType: "ALL",
        workTopics: [],
        workTopicMode: "any",
        excludedWorkTopics: [],
        workType: "ALL",
        adultContent: "ALL",
      },
      planningDisplay: {
        favoriteGroupId: "ALL",
        visitStatus: "ALL",
        sort: "booth",
        density: "informative",
        mediaCount: 0,
      },
      navigationMode: false,
      selectedRecordId: null,
    };
    const projections = event.days.map((day) => {
      const projected = projectEventWorkspace({ ...base, day: day.id });
      return {
        day: day.id,
        filtered: projected.filtered.map(record),
        mapRecords: projected.mapRecords.map(record),
        markers: projected.markers.map((marker) => ({ code: marker.code, records: marker.records.map(record) })),
        slots: projected.slots,
        favorites: projected.favorites,
        dayPlan: projected.dayPlan,
        genreCounts: [...projected.genreCounts.entries()],
        activeFilterDescriptors: projected.activeFilterDescriptors,
      };
    });
    const canonical = (value) => {
      if (Array.isArray(value)) return value.map(canonical);
      if (value && typeof value === "object") return Object.fromEntries(
        Object.keys(value).sort().map((key) => [key, canonical(value[key])]),
      );
      return value;
    };
    const layout = mapValue.layout;
    const projectionBytes = canonical({
      event: {
        id: event.id,
        days: event.days,
        areas: event.areas,
        venueAssignments: event.venueAssignments,
        genres: event.genres,
        dataUpdatedAt: event.dataUpdatedAt,
      },
      map: {
        eventId: mapValue.eventId,
        revision: mapValue.revision,
        sourceName: mapValue.sourceName,
        width: layout.width,
        height: layout.height,
        rows: layout.rows.length,
        slots: layout.rows.reduce((total, row) => total + row.slots.length, 0),
        pillars: layout.pillars.length,
        accessPoints: layout.accessPoints.length,
        landmarks: layout.landmarks.length,
      },
      catalog: {
        generatedAt: catalog.generatedAt,
        circles: catalog.circles,
        placements: catalog.placements,
        records: catalog.records.map(record),
      },
      projections,
    });
    const report = {
      result: "PASS",
      label: options.label ?? "unlabelled",
      workspace,
      node: process.version,
      pin: {
        repository: pin.repository,
        commit: pin.commit,
        fileCount: pin.files.length,
      },
      pinnedInputHashes,
      stagedOutputHashes,
      official: {
        schemaVersion: official.schemaVersion,
        days: officialDays,
        groups: officialDays.reduce((total, day) => total + day.groups, 0),
        codes: officialDays.reduce((total, day) => total + day.codes, 0),
        multiCodeGroups: officialDays.reduce((total, day) => total + day.multiCodeGroups, 0),
      },
      map: {
        eventId: mapValue.eventId,
        revision: mapValue.revision,
        sourceName: mapValue.sourceName,
        width: layout.width,
        height: layout.height,
        rows: layout.rows.length,
        slots: layout.rows.reduce((total, row) => total + row.slots.length, 0),
        pillars: layout.pillars.length,
        accessPoints: layout.accessPoints.length,
        landmarks: layout.landmarks.length,
      },
      catalog: {
        circles: catalog.circles.length,
        placements: catalog.placements.length,
        records: catalog.records.length,
        uniqueCodes: new Set(catalog.records.map((value) => value.code)).size,
      },
      eventData: { id: event.id, days: event.days.length, areas: event.areas.length },
      projectionCounts: projections.map((item) => ({
        day: item.day,
        filtered: item.filtered.length,
        mapRecords: item.mapRecords.length,
        markers: item.markers.length,
        slots: Object.keys(item.slots).length,
      })),
      projectionSha256: hashBytes(Buffer.from(JSON.stringify(projectionBytes))),
    };
    if (!Object.values(pinnedInputHashes).every((value) => value.match)) throw new Error("Pinned input hash verification failed.");
    await writeFile(path.resolve(options.output), `${JSON.stringify(report, null, 2)}\n`);
    console.log(JSON.stringify(report, null, 2));
  } finally {
    await vite.close();
  }
}
